1. Python 客户端远程调用代码 (工控机/外部服务)

          
import requests

url = "http://127.0.0.1:8080/api/v1/predict"

# 上传待检测零件图片
with open("test_part.png", "rb") as f:
    response = requests.post(url, files={"file": f})
    result = response.json()

print(f"检测状态: {result['status']}, 耗时: {result['duration_ms']}ms")
print(f"检出合格目标数量: {result['object_count']}")
for obj in result["objects"]:
    print(f" -> 目标 #{obj['id']} 面积={obj['area']}, 圆度={obj['circularity']}")
